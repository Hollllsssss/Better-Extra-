local function escapePattern(s)
	return (tostring(s):gsub("([%^%$%(%)%%%.%[%]%*%+%-%?])", "%%%1"))
end

function validate(name)
	name = tostring(name or "")
	for i,v in pairs(bannedsymbols) do
		local ok, res = pcall(function()
			return name:gsub(escapePattern(i), v)
		end)
		if ok then name = res end
	end
	local s = string.find(name,"%.txt",1,true) or string.find(name,"%.json",1,true)
	if s then
		local addafter = string.sub(name,s)
		name = string.sub(name,1,s-1)
		name = name:gsub("%.","·")
		name = name..addafter
	else
		name = name:gsub("%.","·")
	end
	return name
end

local function safeLoad(url, name)
    local ok, src = pcall(game.HttpGet, game, url)
    if not ok then
        warn("[TCOE] HTTP GET failed for " .. name .. ": " .. tostring(src))
        return nil
    end
    if not src or src == "" then
        warn("[TCOE] Empty HTTP response for " .. name .. " (" .. url .. ")")
        return nil
    end
    local fn, lerr = loadstring(src)
    if type(fn) ~= "function" then
        warn("[TCOE] loadstring failed for " .. name .. ": " .. tostring(lerr))
        return nil
    end
    local ok2, res = pcall(fn)
    if not ok2 then
        warn("[TCOE] Running " .. name .. " failed: " .. tostring(res))
        return nil
    end
    return res
end

function lib.maketab(name)
    local icon = "solar/widget-bold"

    elseif n == "credit" then icon = "solar/heart-bold"
    elseif n == "moderation & management" then icon = "solar/shield-bold" end

    -- or a second metadata table, and uses the exact AddSection API from Fluent-Modded.

    return tab
end

function lib.maketextbox(text, tab, callback)
	local id = "TCOE_" .. tostring(math.random(100000,999999))
	local lastFired = nil
	local function fire(value)
		if value == nil or value == "" then return end
		if value == lastFired then return end
		lastFired = value
		local ok, err = pcall(callback, value)
		if not ok then
			warn("[TCOE] textbox callback error:", err)
			pcall(function() notify("Input Error", tostring(err)) end)
		end
	end
	local input = sectionFor(tab):AddInput(id, {
		Title = tostring(text),
		Placeholder = tostring(text),
		Default = "",
		Finished = true,
		Callback = fire,
	})
	-- Mobile fallback: some Fluent builds only fire Finished on Enter
	pcall(function()
		local tb = input.TextBox or input.Instance or input.Textbox
		if tb and tb.FocusLost then
			tb.FocusLost:Connect(function()
				fire(tb.Text)
			end)
		end
	end)
	local proxy = {}
	setmetatable(proxy, {
		__index = function(_, key)
			if key == "Text" then
				return input.Value or ""
			elseif key == "PlaceholderText" then
				return input.Placeholder or ""
			end
			return input[key]
		end,
		__newindex = function(_, key, value)
			if key == "Text" then
				pcall(function() input:SetValue(tostring(value or "")) end)
			elseif key == "PlaceholderText" then
				pcall(function() input:SetPlaceholder(tostring(value or "")) end)
			else
				pcall(function() input[key] = value end)
			end
		end,
	})
	return proxy
end

function lib.updatelabel(text, label)
    if not label then return end
    pcall(function()
        if label.SetTitle then label:SetTitle(tostring(text)) return end
        if label.Title then label.Title.Text = tostring(text) return end
        if label.Container then
            for _,v in ipairs(label.Container:GetDescendants()) do
                if v:IsA("TextLabel") and v.Text ~= "" then
                    v.Text = tostring(text)
                    return
                end
            end
        end
    end)
end

end

local uiSizeInputs = {}

local function readSizeInput(id, fallback)
    local inp = uiSizeInputs[id]
    if not inp then return fallback end
    local raw = inp.Value
    if raw == nil and inp.TextBox then raw = inp.TextBox.Text end
    return tonumber(raw) or fallback
end

local function applyUISizes()
    local w = math.clamp(math.floor(readSizeInput("TCOE_UIWidth",  680)), 101, 2000)
    local h = math.clamp(math.floor(readSizeInput("TCOE_UIHeight", 390)), 101, 1500)
    Window.Size = UDim2.fromOffset(w, h)
    Window.Root.Size = UDim2.fromOffset(w, h)
    notify("UI Size", "Applied: " .. w .. " x " .. h)
end

local function applyIconSize(size)
    size = math.clamp(math.floor(tonumber(size) or 56), 16, 100)
    local cg = game:GetService("CoreGui")
    local mg = cg:FindFirstChild("FluentMinimizerGui")
    if not mg then
        for _, g in ipairs(cg:GetChildren()) do
            if g:IsA("ScreenGui") and g.Name:lower():find("minimizer") then
                mg = g; break
            end
        end
    end
    if not mg then return end
    local btn = mg:FindFirstChild("FluentMinimizerBtn", true)
    if not btn then
        for _, d in ipairs(mg:GetDescendants()) do
            if d:IsA("GuiButton") then btn = d; break end
        end
    end
    if btn then
        btn.Size = UDim2.fromOffset(size, size)
        local img = btn:FindFirstChildWhichIsA("ImageLabel", true)
        if img then
            local pad = 6
            img.Size = UDim2.new(1, -(pad * 2), 1, -(pad * 2))
        end
    end
end

function listfilesfixed(directory)
	local s, lf = pcall(function()
		return listfiles(directory)
	end)
	if s and type(lf) == "table" then
		for i, v in pairs(lf) do
			if type(v) == "string" and string.sub(v, 1, 2) == "./" then
				lf[i] = string.sub(v, 3)
			end
		end
	else
		warn("[TCOE] listfiles failed for '"..tostring(directory).."': "..tostring(lf))
		lf = {}
	end
	return lf
end

function getfn(js, first)

			if type(v) == "string" then

			if type(v) == "string" then
				fn[i] = (v:gsub("TheChosenOneBuilds/", ""))
			end
		end
	end
	return fn
end

-- Speed multipliers for collide + sign placement
local COLLIDE_WAIT_MULT = 1.0
local SIGN_WAIT_MULT    = 1.0
local function cwait(t) task.wait((t or 0.1) * COLLIDE_WAIT_MULT) end
local function swait(t) task.wait((t or 0.1) * SIGN_WAIT_MULT) end

local function getGhostBlockers(ghost)
    local blockers = {}
    if not ghost or not ghost:IsA("BasePart") then return blockers end
    local ghostCFrame = ghost.CFrame
    local ghostSize = ghost.Size
    local ghostParts = workspace.Bricks:GetDescendants()
    for _,part in ipairs(ghostParts) do
        if part:IsA("BasePart") and part ~= ghost and part.Parent then
            local rel = ghostCFrame:ToObjectSpace(part.CFrame)
            local rs = rel.Position
            local ps = part.Size
            local half = ghostSize/2
            local phalf = ps/2
            if math.abs(rs.X) < (half.X + phalf.X) and
               math.abs(rs.Y) < (half.Y + phalf.Y) and
               math.abs(rs.Z) < (half.Z + phalf.Z) and
               part.CanCollide == true then
                table.insert(blockers, part)
            end
        end
    end
    return blockers
end

local function makeBlockerNonCollidable(part)
    if not part or not part.Parent or part.CanCollide == false then return true end
    local paint = (localplr.Character and localplr.Character:FindFirstChild("Paint")) or (localplr.Backpack and localplr.Backpack:FindFirstChild("Paint"))
    if not paint then return false end
    local ok = pcall(function()
        if not localplr.Character:FindFirstChild("Paint") then
            paint.Parent = localplr.Character
        end
        if localplr.Character:FindFirstChild("Paint") then
            local args = {
                [1] = part,
                [2] = Enum.NormalId.Top,
                [3] = part.Position + part.Size/2,
                [4] = "material",
                [5] = nil,
                [6] = "collide",
                [7] = ""
            }
            local event = (localplr.Character.Paint:FindFirstChild("origevent") and localplr.Character.Paint.origevent:Invoke(unpack(args))) or localplr.Character.Paint.Script.Event:FireServer(unpack(args))
            return true
        end
    end)
    if not ok then return false end
    local tries = 0
    repeat
        tries = tries + 1
        cwait(0.15)
        if not part.Parent then return true end
    until part.CanCollide == false or tries >= 12
    return part.CanCollide == false
end

local function clearCurrentGhostBlockers()
    if not oldprt or not oldprt.Parent then return true end
    local blockers = getGhostBlockers(oldprt)
    for _,part in ipairs(blockers) do
        makeBlockerNonCollidable(part)
    end
    return true
end

local function clearBlockersForSign(P, signSize, excludeParts)
    excludeParts = excludeParts or {}
    local pad = 0.05
    local probe = Instance.new("Part")
    probe.Anchored = true
    probe.CanCollide = false
    probe.CanQuery = false
    probe.CanTouch = false
    probe.CastShadow = false
    probe.Transparency = 1
    probe.Size = Vector3.new(signSize.X + pad, signSize.Y + pad, signSize.Z + pad)
    probe.CFrame = CFrame.new(P)
    probe.Parent = workspace

    local op = OverlapParams.new()
    op.FilterType = Enum.RaycastFilterType.Include
    op.FilterDescendantsInstances = {workspace.Bricks}
    op.MaxParts = 50

    local blockers = {}
    for _, p in ipairs(workspace:GetPartsInPart(probe, op)) do
        if p ~= probe and p.Parent and p.CanCollide == true and not excludeParts[p] then
            table.insert(blockers, p)
        end
    end
    probe:Destroy()

    if #blockers == 0 then return 0 end

    if localplr.Backpack:FindFirstChild("Paint") and localplr.Character then
        pcall(function() localplr.Backpack.Paint.Parent = localplr.Character end)
    end
    local paint = localplr.Character and localplr.Character:FindFirstChild("Paint")
    if not paint then return 0 end

    for _, blk in ipairs(blockers) do
        pcall(function()
            local args = { blk, Enum.NormalId.Top, blk.Position + blk.Size/2, "material", nil, "collide", "" }
            local ev = paint:FindFirstChild("origevent")
            if ev then ev:Invoke(unpack(args)); return end
            local scr = paint:FindFirstChild("Script")
            if scr and scr:FindFirstChild("Event") then scr.Event:FireServer(unpack(args)) end
        end)
        cwait(0.3)
    end

    return #blockers
end

		anchored = true
	end

		collide = true
	end

	local s,e = pcall(function()
		localplr.Backpack.Build.Parent = localplr.Character
	end)
	local oo = false

	if oldprt and oldprt.Parent then
		clearCurrentGhostBlockers()
	end
	if #cubehistory > 0 and oldprt then
		local allooslol = {}
		for i,childcube2 in pairs(cubehistory) do
			if childcube2 == nil or childcube2.Parent == nil then
				cubehistory[i] = nil
				continue
			elseif oldprt.Size == childcube2.Size then
				for i,v in pairs(normalids) do
					local pos = childcube2.Position+(v[1]*childcube2.Size[v[2]])
					if pos == oldprt.Position then
						oo = {i,childcube2,childcube2.Position+(v[1]*childcube2.Size[v[2]]/2)}
						table.insert(allooslol,{i,childcube2,childcube2.Position+(v[1]*childcube2.Size[v[2]]/2)})
					end
				end
			end
		end
		if #allooslol > 1 and color and oo[2].Color ~= color then
			for i,v in pairs(allooslol) do
				if v[2].Color == color then
					oo = v
				end
			end
		end
		local origposs = pos
		if oo and oo[2] ~= nil and oo[2].Parent ~= nil then
			local args = {
				[1] = oo[2],
				[2] = oo[1],
				[3] = oo[3] or oldprt.Position,
				[4] = "normal"
			}
			built = false
			childcube = nil
			c = 0
			repeat
				c = c + 1
				if localplr.Character:FindFirstChild("Build") then
					local event = (localplr.Character.Build:FindFirstChild("origevent") and localplr.Character.Build.origevent:Invoke(unpack(args))) or localplr.Character.Build.Script.Event:FireServer(unpack(args))
				else
					local s,e = pcall(function()
						localplr.Backpack.Build.Parent = localplr.Character
					end)
				end
				local s,e = pcall(function()
					novel = true
					pos = oo[3] or pos
					if tp then
						localplr.Character.HumanoidRootPart.CFrame = CFrame.new(pos)
					end
				end)
				task.wait(.05)
			until (built == true and childcube) or oo[2] == nil or oo[2].Parent == nil or stopped == true or skipblock == true or c > 200

-- ============================================================
-- FACE_NORMALS + grid helpers + sign placement
-- ============================================================
local FACE_NORMALS = {
    Front  = Vector3.new(0, 0, -1),
    Back   = Vector3.new(0, 0, 1),
    Left   = Vector3.new(-1, 0, 0),
    Right  = Vector3.new(1, 0, 0),
    Top    = Vector3.new(0, 1, 0),
    Bottom = Vector3.new(0, -1, 0),
}

local function isAtGridXZ(pos)
    local function nearGrid(n)
        local k = math.round((n - 2) / 4)
        return math.abs(n - (k * 4 + 2)) < 0.1
    end
    return nearGrid(pos.X) and nearGrid(pos.Z)
end

local function computeBlockPlacement(P, faceName, signSize)
    local n = FACE_NORMALS[faceName] or Vector3.new(0, 0, -1)
    local sx, sy, sz = signSize.X, signSize.Y, signSize.Z
    local signThick = math.min(sx, sy, sz)
    local signY = sy
    local signWidth
    if math.abs(sx - signThick) < 0.01 then
        signWidth = sz
    elseif math.abs(sz - signThick) < 0.01 then
        signWidth = sx
    else
        signWidth = math.max(sx, sz)
    end
    local blockHeight
    if signY >= 1 then
        blockHeight = signY + 1
    else
        blockHeight = 1
    end
    local blockDepth = signWidth
    local blockSize
    if faceName == "Front" or faceName == "Back" then
        blockSize = Vector3.new(signWidth, blockHeight, blockDepth)
    elseif faceName == "Left" or faceName == "Right" then
        blockSize = Vector3.new(blockDepth, blockHeight, signWidth)
    else
        blockSize = Vector3.new(signWidth, blockHeight, blockDepth)
    end
    local blockCenter = P - n * (signThick / 2 + blockDepth / 2)
    return blockCenter, blockSize
end

local function applySignColorMaterial(signPart, signData, materialsTable)
    if not signPart or not signPart.Parent then return end
    if not signPart:FindFirstChildOfClass("SurfaceGui") then return end

    local wantColor = nil
    if type(signData.c) == "table" and #signData.c == 3 then
        wantColor = Color3.fromRGB(signData.c[1], signData.c[2], signData.c[3])
    end
    local wantMatName = nil
    if signData.o and materialsTable then
        local ok, mat = pcall(function() return Enum.Material[signData.o] end)
        if ok and mat and materialsTable[mat] then
            wantMatName = materialsTable[mat]
        end
    end
    local colorSame = (not wantColor) or (wantColor == signPart.Color)
    local matSame = (not wantMatName) or (materialsTable and materialsTable[signPart.Material] == wantMatName)
    if colorSame and matSame then return end

    if localplr.Backpack:FindFirstChild("Paint") and localplr.Character then
        pcall(function() localplr.Backpack.Paint.Parent = localplr.Character end)
    end
    local p = localplr.Character and localplr.Character:FindFirstChild("Paint")
    if not p then return end

    local faceId = Enum.NormalId[signData.signFace or "Front"] or Enum.NormalId.Front
    local mode, texArg
    if wantColor and wantMatName then
        mode = "both \u{1F91D}"
        texArg = wantMatName
    elseif wantColor then
        mode = "color"
        texArg = "tiles"
    elseif wantMatName then
        mode = "material"
        texArg = wantMatName
    end
    local args = { signPart, faceId, signPart.Position, mode, wantColor, texArg, "" }
    pcall(function()
        local ev = p:FindFirstChild("origevent")
        if ev then ev:Invoke(unpack(args)); return end
        local scr = p:FindFirstChild("Script")
        if scr and scr:FindFirstChild("Event") then scr.Event:FireServer(unpack(args)) end
    end)
    swait(0.4)
end

			elseif sign:FindFirstChild("Script") and sign.Script:FindFirstChild("Event") then sign.Script.Event:FireServer(unpack(args)) end
		end)
		swait(0.8)
		placedSign = findNearbySign(5)
		if placedSign and signText ~= "" then
			applySignText(placedSign, signText)
			swait(0.4)
		end

		if signData.signNoLeg and placedSign then
			local myModel = workspace.Bricks:FindFirstChild(localplr.Name)
			if myModel then
				local toDelete = {}
				for _, p in ipairs(myModel:GetChildren()) do
					if p:IsA("BasePart") and p ~= placedSign then
						local hasGui = false
						for _, sg in ipairs(p:GetChildren()) do
							if sg:IsA("SurfaceGui") then hasGui = true; break end
						end
						if not hasGui then
							local dx = math.abs(p.Position.X - placedSign.Position.X)
							local dz = math.abs(p.Position.Z - placedSign.Position.Z)
							local dy = placedSign.Position.Y - p.Position.Y
							if dx < 0.3 and dz < 0.3 and dy > 0.5 and dy < 3.5 then
								table.insert(toDelete, p)
							end
						end
					end
				end
				for _, p in ipairs(toDelete) do
					deletePartViaTool(p)
					swait(0.4)
				end
			end
		end
	else
		if parentPart and parentPart.Parent then
			local viewPos = P + n * 6 + Vector3.new(0, 1, 0)
			teleport(viewPos, P)
			swait(0.2)
			clearBlockersForSign(P, Vector3.new(sw, sh, st), {parentPart})
			local args = { parentPart, faceId, blockFacePos, "normal" }
			pcall(function()
				if sign:FindFirstChild("origevent") then sign.origevent:Invoke(unpack(args))
				elseif sign:FindFirstChild("Script") and sign.Script:FindFirstChild("Event") then sign.Script.Event:FireServer(unpack(args)) end
			end)
			swait(0.7)
			placedSign = findNearbySign(3)
			if placedSign and signText ~= "" then
				applySignText(placedSign, signText)
				swait(0.4)
			end
		else
			local viewPos = P + n * 6 + Vector3.new(0, 1, 0)
			teleport(viewPos, P)

			clearBlockersForSign(P, Vector3.new(sw, sh, st), {})

			local blockCenter, blockSize = computeBlockPlacement(P, faceName, Vector3.new(sw, sh, st))
			local tempBlock = buildTempBlock(blockCenter, blockSize)
			swait(0.5)

			if not tempBlock then
				notify("Sign", "Could not build temp block")
				if oldprt then oldprt:Destroy(); oldprt = nil end
				return nil
			end

			local args = { tempBlock, faceId, blockFacePos, "normal" }
			pcall(function()
				if sign:FindFirstChild("origevent") then sign.origevent:Invoke(unpack(args))
				elseif sign:FindFirstChild("Script") and sign.Script:FindFirstChild("Event") then sign.Script.Event:FireServer(unpack(args)) end
			end)
			swait(0.7)
			placedSign = findNearbySign(3)
			if placedSign and signText ~= "" then
				applySignText(placedSign, signText)
				swait(0.4)
			end

			for _ = 1, 12 do
				if not tempBlock or not tempBlock.Parent then break end
				deletePartViaTool(tempBlock)
				swait(0.5)
			end
			if oldprt then oldprt:Destroy(); oldprt = nil end
		end
	end

	if placedSign and placedSign.Parent and wantCollide == false then
		local paint = localplr.Backpack:FindFirstChild("Paint") or (localplr.Character and localplr.Character:FindFirstChild("Paint"))
		if paint then
			if not localplr.Character:FindFirstChild("Paint") then
				pcall(function() paint.Parent = localplr.Character end)
			end
			local p = localplr.Character:FindFirstChild("Paint")
			if p then
				local args = { placedSign, Enum.NormalId.Top, placedSign.Position + placedSign.Size/2, "material", nil, "collide", "" }
				pcall(function()
					local ev = p:FindFirstChild("origevent")
					if ev then ev:Invoke(unpack(args)); return end
					local scr = p:FindFirstChild("Script")
					if scr and scr:FindFirstChild("Event") then scr.Event:FireServer(unpack(args)) end
				end)
				swait(0.3)
			end
		end
	end

	if placedSign and placedSign.Parent then
		applySignColorMaterial(placedSign, signData, materialsTable)
	end

	return placedSign
end

function createpartpos(pos,col)
	if typeof(pos) == "CFrame" then
		pos = pos.Position
	end
	local p = Instance.new("Part")
	p.Shape = Enum.PartType.Ball
	p.Anchored = true
	p.CanCollide = false
	p.CastShadow = false
	p.CanQuery = false
	p.Color = col
	p.Transparency = .5
	p.Size = Vector3.new(3,3,3)
	p.CFrame = CFrame.new(pos)
	p.Parent = workspace
end
function createpartrepl(pos,bsize,col,mat,transp,anch,collide,sprays)
	if typeof(pos) == "Vector3" then
		pos = CFrame.new(pos)
	end
	local p = Instance.new("Part")
	oldprt = p
	p.Anchored = anch or true
	p.CanCollide = collide
	if p.CanCollide == nil then
		p.CanCollide = false
	end
	p.CastShadow = false
	p.CanQuery = false
	p.Color = col
	p.Transparency = transp or .5
	p.Material = mat
	if bsize ~= nil then
		pos = CFrame.new((pos.X + (bsize.X/2))-.5,(pos.Y + (bsize.Y/2))-.5,(pos.Z + (bsize.Z/2))-.5) * pos.Rotation
	end
	p.Size = bsize or Vector3.new(mult,mult,mult)
	p.CFrame = pos
	if sprays then
		for i,v in pairs(sprays) do
			local face = Enum.NormalId[v[1]]
			local image = v[2]
			local txt = v[3]
			local surfaceui = Instance.new("SurfaceGui")
			surfaceui.Face = face
			surfaceui.SizingMode = Enum.SurfaceGuiSizingMode.PixelsPerStud
			surfaceui.PixelsPerStud = 50
			local out,count = string.gsub(txt,"#","l")
			if count == string.len(txt) then
				local img = Instance.new("ImageLabel")
				img.Image = image
				img.BackgroundTransparency = 1
				img.Size = UDim2.new(1,0,1,0)
				img.Parent = surfaceui
			else
				local textlabel = Instance.new("TextLabel")
				textlabel.Text = txt
				textlabel.BackgroundTransparency = 1
				textlabel.TextScaled = true
				textlabel.TextColor3 = Color3.fromRGB(255,255,255)
				textlabel.Font = Enum.Font.FredokaOne
				textlabel.Size = UDim2.new(1,0,1,0)
				textlabel.Parent = surfaceui
			end
			surfaceui.Parent = p
		end
	end
	p.Parent = workspace
	return p
end

function saveblock(bl)
	local blockdata = {}
	if bl:IsA("BasePart") then
		local p = bl.Position
		local pt = {bl.Position.X,bl.Position.Y,bl.Position.Z}
		if (bl.CFrame - bl.Position) ~= CFrame.new() then
			blockdata.p = {bl.CFrame:GetComponents()}
		else
			blockdata.p = pt
		end
		blockdata.c = {math.round(bl.Color.R*255),math.round(bl.Color.G*255),math.round(bl.Color.B*255)}
		blockdata.a = bl.Anchored
		blockdata.cc = bl.CanCollide
		if bl.Size.X ~= mult or bl.Size.Y ~= mult or bl.Size.Z ~= mult then
			blockdata.p[1] = (blockdata.p[1] - (bl.Size.X/2))+.5
			blockdata.p[2] = (blockdata.p[2] - (bl.Size.Y/2))+.5
			blockdata.p[3] = (blockdata.p[3] - (bl.Size.Z/2))+.5
			blockdata.s = {bl.Size.X,bl.Size.Y,bl.Size.Z}
		else
		end
		blockdata.m = materials[bl.Material] or "plastic"
		blockdata.o = bl.Material.Name
		blockdata.sp = {}
		for i,v in pairs(bl:GetChildren()) do
			if v.Name == "Spray" then
				local escaped = (v.Label.Text:gsub('"','\\"'))
				table.insert(blockdata.sp,{v.Face.Name,v.Image.Image,escaped})
			end
		end
	end
	return blockdata
end

isSignPanel = function(part)
	if not part:IsA("BasePart") then return false, nil, nil end
	for _, sg in ipairs(part:GetChildren()) do
		if sg:IsA("SurfaceGui") then
			local tb = sg:FindFirstChildWhichIsA("TextBox", true)
			if tb then return true, sg.Face.Name, tb.Text or "" end
		end
	end
	return false, nil, nil
end

findSignLeg = function(panel, model)
	for _, p in ipairs(model:GetChildren()) do
		if p:IsA("BasePart") and p ~= panel then
			local hasGui = false
			for _, sg in ipairs(p:GetChildren()) do if sg:IsA("SurfaceGui") then hasGui = true; break end end
			if not hasGui then
				local dx = math.abs(p.Position.X - panel.Position.X)
				local dz = math.abs(p.Position.Z - panel.Position.Z)
				local dy = panel.Position.Y - p.Position.Y
				if dx < 0.1 and dz < 0.1 and dy > 0.5 and dy < 3.5 and p.Size.Y <= 2.5 then return p end
			end
		end
	end
	return nil
end

findParentPart = function(panel, model)
	if not panel or not panel.Parent then return nil end
	local ok, face = isSignPanel(panel)
	if not ok then return nil end
	local n = FACE_NORMALS[face] or Vector3.new(0,0,-1)
	local signThick = math.min(panel.Size.X, panel.Size.Y, panel.Size.Z)
	local contactPos = panel.Position - n * (signThick / 2)
	for _, p in ipairs(model:GetChildren()) do
		if p:IsA("BasePart") and p ~= panel then
			local rel = p.CFrame:PointToObjectSpace(contactPos)
			local h = p.Size / 2
			if math.abs(rel.X) <= h.X + 0.05 and math.abs(rel.Y) <= h.Y + 0.05 and math.abs(rel.Z) <= h.Z + 0.05 then
				return p
			end
		end
	end
	return nil
end

lib.makebutton("Load Save",bsaltab,function()
	if not selectedbuild then return end
	local build = getbuild()
	if not build then return end
	if oldprt then oldprt:Destroy() end

	local normalParts, signParts = {}, {}
	for _, v in ipairs(build) do
		if v.isSign then table.insert(signParts, v) else table.insert(normalParts, v) end
	end

	bulkLoading = true
	for _, v in ipairs(normalParts) do
		if stopped then break end
		local posses = v.p or v.pos
		if posses then
			local args = {}
			args[1] = CFrame.new(posses[1],posses[2],posses[3]).Position + gizmoOffset
			args[2] = v.m or v.mat
			args[3] = Color3.fromRGB(table.unpack(v.c or v.color))
			if v.s or v.size then args[5] = Vector3.new(table.unpack(v.s or v.size)) end
			createpartrepl(args[1],args[5],args[3],swappedmaterials[v.m or v.mat])
			args[7] = v.o or v.origmat
			args[8] = v.sp or v.sprayed
			args[9] = v.a or v.anchored
			args[10] = v.cc
			if args[10] == nil then args[10] = v.collide end
			buildblock(args[1],args[2],args[3],nil,args[5],nil,args[7],args[8],args[9],args[10])
		end
	end
	bulkLoading = false

	local idMap = {}
	for _, v in ipairs(build) do if v.id then idMap[v.id] = v end end
	local function getDepth(v)
		local d, pid, guard = 1, v.parentId, 0
		while pid and idMap[pid] and guard < 30 do
			d = d + 1
			pid = idMap[pid].parentId
			guard = guard + 1
		end
		return d
	end
	table.sort(signParts, function(a, b) return getDepth(a) < getDepth(b) end)

	local idToPart = {}
	for _, v in ipairs(signParts) do
		if stopped then break end
		local pCopy = v.p or v.pos
		if pCopy and #pCopy == 12 then
			local cf = CFrame.new(table.unpack(pCopy)) + gizmoOffset
			v._worldPos = cf.Position
		elseif pCopy and #pCopy == 3 then
			local size = v.s or v.size or {mult,mult,mult}
			v._worldPos = Vector3.new(
				pCopy[1] + size[1]/2 - 0.5 + gizmoOffset.X,
				pCopy[2] + size[2]/2 - 0.5 + gizmoOffset.Y,
				pCopy[3] + size[3]/2 - 0.5 + gizmoOffset.Z
			)
		end
		local parentPart = nil
		if v.parentId and idToPart[v.parentId] then parentPart = idToPart[v.parentId] end
		local signPart = placeSign(v, parentPart, materials)
		if signPart and v.id then idToPart[v.id] = signPart end
	end

	stopped = false
end)

function optimizeLoadOrder(build)
	if not build or #build < 2 then
		return build
	end

	local nodes = {}
	local groups = {}

	local function getpos(v)
		local p = v.p or v.pos
		if not p then return nil end
		return Vector3.new(p[1],p[2],p[3])
	end

	local function getsize(v)
		local size = v.s or v.size
		if size then
			return Vector3.new(size[1],size[2],size[3])
		end
		return Vector3.new(mult,mult,mult)
	end

	local function numkey(n)
		return tostring(math.round(n * 10000) / 10000)
	end

	local function poskey(p)
		return numkey(p.X).."|"..numkey(p.Y).."|"..numkey(p.Z)
	end

	local function sizekey(size)
		return numkey(size.X).."|"..numkey(size.Y).."|"..numkey(size.Z)
	end

	for i,v in ipairs(build) do
		local pos = getpos(v)
		if pos then
			local size = getsize(v)
			local node = {data=v,index=i,pos=pos,size=size}
			table.insert(nodes,node)
			local sk = sizekey(size)
			groups[sk] = groups[sk] or {}
			groups[sk][poskey(pos)] = node
		end
	end

	local dirs = {
		Vector3.new(1,0,0),Vector3.new(-1,0,0),
		Vector3.new(0,1,0),Vector3.new(0,-1,0),
		Vector3.new(0,0,1),Vector3.new(0,0,-1)
	}

	local ordered = {}
	local visited = {}

	for _,node in ipairs(nodes) do
		if not visited[node] then
			local queue = {node}
			local qhead = 1
			visited[node] = true
			local group = groups[sizekey(node.size)]

			while qhead <= #queue do
				local current = queue[qhead]
				qhead = qhead + 1
				table.insert(ordered,current.data)

				for _,dir in ipairs(dirs) do
					local np = current.pos + Vector3.new(
						dir.X * current.size.X,
						dir.Y * current.size.Y,
						dir.Z * current.size.Z
					)
					local nextnode = group[poskey(np)]
					if nextnode and not visited[nextnode] then
						visited[nextnode] = true
						table.insert(queue,nextnode)
					end
				end
			end
		end
	end

	return ordered
end

local function roundTransformNumber(n)
	return math.round(n * 100) / 100
end

local updatingTransformFields = false

updateTransformFields = function()
	if updatingTransformFields then return end
	updatingTransformFields = true

	local ok, err = pcall(function()
		local worldPosition = gizmoPivotCenter and (gizmoPivotCenter + gizmoOffset) or gizmoOffset
		if positionBoxes.X then positionBoxes.X.Text=tostring(roundTransformNumber(worldPosition.X)) end
		if positionBoxes.Y then positionBoxes.Y.Text=tostring(roundTransformNumber(worldPosition.Y)) end
		if positionBoxes.Z then positionBoxes.Z.Text=tostring(roundTransformNumber(worldPosition.Z)) end
	end)

	updatingTransformFields = false
	if not ok then warn("Better Extra Transform UI update error:", err) end
end

getPreviewCenter = function()
	if gizmoPivotCenter then return gizmoPivotCenter + gizmoOffset end
	if not prttable or #prttable==0 then return nil end
	local minv=Vector3.new(math.huge,math.huge,math.huge)
	local maxv=Vector3.new(-math.huge,-math.huge,-math.huge)
	local valid=false
	for _,v in ipairs(prttable) do
		if v and v.Parent and v:IsA("BasePart") then
			valid=true
			local half=v.Size/2
			local p=v.Position
			minv=Vector3.new(math.min(minv.X,p.X-half.X),math.min(minv.Y,p.Y-half.Y),math.min(minv.Z,p.Z-half.Z))
			maxv=Vector3.new(math.max(maxv.X,p.X+half.X),math.max(maxv.Y,p.Y+half.Y),math.max(maxv.Z,p.Z+half.Z))
		end
	end
	if not valid then return nil end
	return (minv+maxv)/2
end

local function setPositionAxis(axis,value)
	value=tonumber(value)
	if not value then return end
    value = math.floor(value * 2 + 0.5) / 2
	local base = gizmoPivotCenter or Vector3.new(0,0,0)
	local worldPosition = gizmoPivotCenter and (gizmoPivotCenter + gizmoOffset) or gizmoOffset
	if axis=="X" then worldPosition=Vector3.new(value,worldPosition.Y,worldPosition.Z)
	elseif axis=="Y" then worldPosition=Vector3.new(worldPosition.X,value,worldPosition.Z)
	else worldPosition=Vector3.new(worldPosition.X,worldPosition.Y,value) end
	gizmoOffset=worldPosition-base
	applyPositionOffsetToPreview()
	if offsetgizmo then createAxisGizmo() end
end

local function createArrowAxis(folder,key,direction,color,length,thickness,headLength)
	local center=getPreviewCenter()
	if not center then return end
	local rod=makeGizmoPart(folder,"Axis_"..key,Vector3.new(thickness,thickness,length),CFrame.lookAt(center+direction*(length/2),center+direction*(length/2)+direction),color)
	rod:SetAttribute("GizmoAxis",key)
	local side=direction:Cross(Vector3.new(0,1,0))
	if side.Magnitude<.01 then side=direction:Cross(Vector3.new(1,0,0)) end
	side=side.Unit
	local up=side:Cross(direction).Unit
	local tip=center+direction*(length+headLength*.45)
	local wingCenter=tip-direction*(headLength*.45)
	for n,sgn in ipairs({1,-1}) do
		local wingDir=(-direction*.72+up*sgn*.72).Unit
		local wing=makeGizmoPart(folder,"Arrow_"..key.."_"..n,Vector3.new(thickness*.8,thickness*.8,headLength),CFrame.lookAt(wingCenter+up*sgn*(headLength*.32),wingCenter+up*sgn*(headLength*.32)+wingDir),color)
		wing:SetAttribute("GizmoAxis",key)
	end
end

createAxisGizmo = function()
	destroyOffsetGizmo()
	if not prttable or #prttable==0 then return end
	local center=getPreviewCenter()
	if not center then return end
	local folder=Instance.new("Folder")
	folder.Name="WholeBuildOffsetGizmo"
	folder.Parent=workspace
	offsetgizmo=folder
	local extent=6
	for _,v in ipairs(prttable) do if v and v.Parent and v:IsA("BasePart") then extent=math.max(extent,v.Size.X,v.Size.Y,v.Size.Z) end end
	local length=math.clamp(extent*1.5,8,20)
	local thickness=math.clamp(length*.055,.3,.75)
	local headLength=math.clamp(length*.28,1.5,3.5)
	createArrowAxis(folder,"X",Vector3.new(1,0,0),Color3.fromRGB(220,90,90),length,thickness,headLength)
	createArrowAxis(folder,"Y",Vector3.new(0,1,0),Color3.fromRGB(100,210,120),length,thickness,headLength)
	createArrowAxis(folder,"Z",Vector3.new(0,0,1),Color3.fromRGB(100,150,225),length,thickness,headLength)

	local uis=game:GetService("UserInputService")
	local camera
	local axisDirections={X=Vector3.new(1,0,0),Y=Vector3.new(0,1,0),Z=Vector3.new(0,0,1)}
	local function rayHit(axis,screenPosition,point)
		camera=workspace.CurrentCamera
		if not camera then return end
		local ray=camera:ScreenPointToRay(screenPosition.X,screenPosition.Y)
		local normal=axis:Cross(camera.CFrame.LookVector)
		if normal.Magnitude<.001 then normal=axis:Cross(camera.CFrame.RightVector) end
		if normal.Magnitude<.001 then return end
		normal=normal.Unit
		local den=ray.Direction:Dot(normal)
		if math.abs(den)<.0001 then return end
		local t=(point-ray.Origin):Dot(normal)/den
		if t<0 then return end
		return ray.Origin+ray.Direction*t
	end
	local function pointToSegmentDistance(point,a,b)
		local ab=b-a
		local denom=ab:Dot(ab)
		if denom<.001 then return (point-a).Magnitude end
		local t=math.clamp((point-a):Dot(ab)/denom,0,1)
		return (point-(a+ab*t)).Magnitude
	end
	local function axisFromScreenPosition(screenPosition)
		camera=workspace.CurrentCamera
		local centerNow=getPreviewCenter()
		if not camera or not centerNow then return end
		local p2=Vector2.new(screenPosition.X,screenPosition.Y)
		local isTouch=uis:GetLastInputType()==Enum.UserInputType.Touch
		local pickRadius=isTouch and 48 or 26
		local bestAxis=nil
		local bestDistance=pickRadius
		for _,key in ipairs({"X","Y","Z"}) do
			local direction=axisDirections[key]
			local a=camera:WorldToViewportPoint(centerNow)
			local b=camera:WorldToViewportPoint(centerNow+direction*(length+headLength+1.5))
			if a.Z>0 or b.Z>0 then
				local dist=pointToSegmentDistance(p2,Vector2.new(a.X,a.Y),Vector2.new(b.X,b.Y))
				if dist<bestDistance then
					bestDistance=dist
					bestAxis=direction
				end
			end
		end
		return bestAxis
	end
	local function beginDrag(axis,pos,touch)
		local centerNow=getPreviewCenter()
		if not centerNow then return end
		local hit=rayHit(axis,pos,centerNow)
		if not hit then return end
		gizmoDragging=true; gizmoAxis=axis; gizmoTouchInput=touch
		gizmoDragStartOffset=gizmoOffset; gizmoDragStartHit=hit; gizmoDragStartCenter=centerNow
		gizmoDragStartCFrames={}
		for i,v in ipairs(prttable or {}) do if v and v.Parent then gizmoDragStartCFrames[i]=v.CFrame end end
		gizmoDragStartGizmoCFrames={}
		for _,v in ipairs(folder:GetChildren()) do if v:IsA("BasePart") then gizmoDragStartGizmoCFrames[v]=v.CFrame end end
	end
	local function updateDrag(pos)
		if not gizmoDragging or not gizmoAxis then return end
		local hit=rayHit(gizmoAxis,pos,gizmoDragStartCenter)
		if not hit then return end
		local delta=(hit-gizmoDragStartHit):Dot(gizmoAxis)
		local step=1
		local movement=gizmoAxis*(math.round(delta/step)*step)
		gizmoOffset=gizmoDragStartOffset+movement
		for i,v in ipairs(prttable or {}) do
			local base=gizmoDragStartCFrames[i]
			if v and v.Parent and base then v.CFrame=base+movement end
		end
		for v,base in pairs(gizmoDragStartGizmoCFrames) do if v and v.Parent then v.CFrame=base+movement end end
		updateTransformFields()
	end
	table.insert(gizmoConnections,uis.InputBegan:Connect(function(input,processed)
		if processed or not offsetgizmo then return end
		local pos; local touch=nil
		if input.UserInputType==Enum.UserInputType.MouseButton1 then pos=Vector2.new(mouse.X,mouse.Y)
		elseif input.UserInputType==Enum.UserInputType.Touch then pos=input.Position; touch=input
		else return end
		local axis=axisFromScreenPosition(pos)
		if axis then beginDrag(axis,pos,touch) end
	end))
	table.insert(gizmoConnections,uis.InputChanged:Connect(function(input)
		if not gizmoDragging then return end
		if input.UserInputType==Enum.UserInputType.MouseMovement then updateDrag(Vector2.new(input.Position.X,input.Position.Y))
		elseif input.UserInputType==Enum.UserInputType.Touch and input==gizmoTouchInput then updateDrag(Vector2.new(input.Position.X,input.Position.Y)) end
	end))
	table.insert(gizmoConnections,uis.InputEnded:Connect(function(input)
		if input.UserInputType==Enum.UserInputType.MouseButton1 or (input.UserInputType==Enum.UserInputType.Touch and input==gizmoTouchInput) then
			gizmoDragging=false; gizmoAxis=nil; gizmoTouchInput=nil
			updateTransformFields()
		end
	end))
end

function makegear(gearname,gearid,save,changename)
	deletegear(gearid)
	if changename == nil then
		changename = true
	end
	if changename then
	local foundcopy = false
	for i,v in pairs(currgears) do
		if v[3] == gearname then
			foundcopy = true
		end
	end
	if foundcopy then
		local n = 1
		local gearname2 = gearname
		repeat
			local foundcopy2 = false
			gearname2 = gearname.." "..tostring(n)
			for i,v in pairs(currgears) do
				if v[3] == gearname2 then
					foundcopy2 = true
				end
			end
			n = n + 1
			if not foundcopy2 then
				foundcopy = false
			end
			task.wait()
		until foundcopy == false
		gearname = gearname2
	end
	end
	local gbutton = lib.makebutton("Copy "..gearname,geartab,function()
		setclipboard(gearcopybefore..tostring(gearid))
	end)
	local gbutton2 = lib.makebutton("Say "..gearname,geartab,function()
		say(gearcopybefore..tostring(gearid),"system")
	end)
	if save then
		currgears[gearid] = {gbutton,gbutton2,gearname,gearid}
		savegears()
	end
end
function deletegear(gearid)
	if currgears[gearid] then
		local v = currgears[gearid]
		pcall(function()
			if type(v) == "table" then
				if type(v[1]) == "table" and v[1].Destroy then v[1]:Destroy() end
				if type(v[2]) == "table" and v[2].Destroy then v[2]:Destroy() end
			end
		end)
		currgears[gearid] = nil
		savegears()
	end
end

	else
		cn = "admin"
	end

	if muted then
		cn = "hidden"

			mdata.Text = ""
		end
	end

	if string.sub(mdata.Text,1,1) == ";" then
		cmd = string.sub(mdata.Text,2,string.find(mdata.Text," ") or nil)
		if on and spychat then
			cn = "hidden"
			hidden = true
		else
			mdata.Text = ""
		end
	end
	if plr == game.Players.LocalPlayer and joincmds[cmd:lower()] then
		for i,v in pairs(joincmds) do
			if localplr.PlayerGui:FindFirstChild(v) then
				localplr.PlayerGui[v].Enabled = i == cmd:lower()
			end
		end

		if plr:GetAttribute("IQ") >= 200 then
			iq = "genius \u{1F9E0}"
			cn = "iqgenius"
		elseif plr:GetAttribute("IQ") <= 50 then
			iq = "dumb \u{1F92A}"
			cn = "iqdumb"
		end
	end
