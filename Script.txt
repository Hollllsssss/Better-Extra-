local function escapePattern(s)
	return (tostring(s):gsub("([%^%$%(%)%%%.%[%]%*%+%-%?])", "%%%1"))
end

function validate(name)
	name = tostring(name or "")
	for i,v in pairs(bannedsymbols) do
		local ok, res = pcall(function()
			return name:gsub(escapePattern(i), v)
		end)
		if ok then name = res end
	end
	local s = string.find(name,"%.txt",1,true) or string.find(name,"%.json",1,true)
	if s then
		local addafter = string.sub(name,s)
		name = string.sub(name,1,s-1)
		name = name:gsub("%.","·")
		name = name..addafter
	else
		name = name:gsub("%.","·")
	end
	return name
end

local function safeLoad(url, name)
    local ok, src = pcall(game.HttpGet, game, url)
    if not ok then
        warn("[TCOE] HTTP GET failed for " .. name .. ": " .. tostring(src))
        return nil
    end
    if not src or src == "" then
        warn("[TCOE] Empty HTTP response for " .. name .. " (" .. url .. ")")
        return nil
    end
    local fn, lerr = loadstring(src)
    if type(fn) ~= "function" then
        warn("[TCOE] loadstring failed for " .. name .. ": " .. tostring(lerr))
        return nil
    end
    local ok2, res = pcall(fn)
    if not ok2 then
        warn("[TCOE] Running " .. name .. " failed: " .. tostring(res))
        return nil
    end
    return res
end

    -- Rebrand visible "ZyronX" text (splash/title strings) to "Poopo Hub".
    -- Only touches string literals, skips anything path/URL-like, and falls back if it would not compile.
    do
        local function brand(text)
            if text:find("/", 1, true) or text:find("\\", 1, true) then return text end
            local function pick(m) return (m == m:upper()) and "POOPO HUB" or "Poopo Hub" end
            text = text:gsub("[Zz][Yy][Rr][Oo][Nn]%s?[Xx]%s?[Uu][Ii]", pick)
            text = text:gsub("[Zz][Yy][Rr][Oo][Nn]%s?[Xx]", pick)
            return text
        end
        local function rebrand(src)
            local out, i, n = {}, 1, #src
            while i <= n do
                local s0 = src:find("[%-\"'%[]", i)
                if not s0 then out[#out + 1] = src:sub(i); break end
                if s0 > i then out[#out + 1] = src:sub(i, s0 - 1) end
                i = s0
                local c = src:sub(i, i)
                if c == "-" then
                    if src:sub(i, i + 1) == "--" then
                        local lb = src:match("^%-%-%[(=*)%[", i)
                        local e
                        if lb then
                            local _, ce = src:find("]" .. lb .. "]", i + 2, true)
                            e = ce or n
                        else
                            e = src:find("\n", i, true) or n
                        end
                        out[#out + 1] = src:sub(i, e)
                        i = e + 1
                    else
                        out[#out + 1] = c
                        i = i + 1
                    end
                elseif c == "[" then
                    local eqs = src:match("^%[(=*)%[", i)
                    if eqs then
                        local open, close = "[" .. eqs .. "[", "]" .. eqs .. "]"
                        local _, ce = src:find(close, i + #open, true)
                        if ce then
                            out[#out + 1] = open .. brand(src:sub(i + #open, ce - #close)) .. close
                            i = ce + 1
                        else
                            out[#out + 1] = src:sub(i)
                            break
                        end
                    else
                        out[#out + 1] = c
                        i = i + 1
                    end
                else
                    local j = i + 1
                    while j <= n do
                        local d = src:sub(j, j)
                        if d == "\\" then j = j + 2
                        elseif d == c or d == "\n" then break
                        else j = j + 1 end
                    end
                    if src:sub(j, j) == c then
                        out[#out + 1] = c .. brand(src:sub(i + 1, j - 1)) .. c
                        i = j + 1
                    else
                        out[#out + 1] = src:sub(i, j - 1)
                        i = j
                    end
                end
            end
            return table.concat(out)
        end
        local okRb, renamed = pcall(rebrand, source)
        if okRb and type(renamed) == "string" and renamed ~= source and type(loadstring(renamed)) == "function" then
            source = renamed
        end
    end

    end
    local ran, library = pcall(chunk)
    if not ran then

        return nil
    end
    return library
end

local pageCache = {}
function lib.setpage(tab, pageName, sectionName)
    if not tab then return end
    pageCache[tab] = pageCache[tab] or {}
    local key = tostring(pageName or "Main") .. "|" .. tostring(sectionName or pageName or "General")
    if not pageCache[tab][key] then
        local page
        for _, existing in ipairs(tab.Pages or {}) do
            if existing.PageName == tostring(pageName or "Main") then
                page = existing
                break
            end
        end
        page = page or tab:CreatePage(tostring(pageName or "Main"))
        pageCache[tab][key] = page:CreateSection(tostring(sectionName or pageName or "General"))
    end
    tabSections[tab] = pageCache[tab][key]
end

function lib.maketextbox(text, tab, callback, live)
    local id = controlId(text, tab)
    local lastFired
    local lastSet
    local function fire(value)
        if value == nil or value == "" or value == lastFired then return end
        lastFired = value
        local ok, err = pcall(callback, value)
        if not ok then
            warn("[TCOE] textbox callback error:", err)
            pcall(function() notify("Input Error", tostring(err), "Error") end)
        end
    end
    local input = sectionFor(tab):AddTextbox(id, tostring(text), fire, {
        Title = tostring(text),
        SubmitOnEnter = true,
        AllowEmpty = true,
    })
    if live then
        -- auto-apply while typing (no Enter needed), debounced like the Settings/Config search boxes
        local box = input and input.Instance
        if box then
            local token = 0
            box:GetPropertyChangedSignal("Text"):Connect(function()
                local v = box.Text
                if v == lastSet then return end -- text written by the script itself, not by the user
                token = token + 1
                local mine = token
                task.delay(0.4, function()
                    if mine == token and box.Parent and box.Text == v then fire(v) end
                end)
            end)
            box.FocusLost:Connect(function()
                token = token + 1
                local v = box.Text
                if v ~= lastSet then fire(v) end
            end)
        end
    end
    local proxy = {}
    setmetatable(proxy, {
        __index = function(_, key)
            if key == "Text" then return input and input.Get and input.Get() or "" end
            if key == "Value" then return input and input.Get and input.Get() or "" end
            if key == "PlaceholderText" then return tostring(text) end
            if key == "TextBox" then return input and input.TextBox end
            return input and input[key]
        end,
        __newindex = function(_, key, value)
            if key == "Text" or key == "Value" then
                lastSet = tostring(value or "")
                lastFired = nil
                if input and input.Set then input:Set(lastSet) end
            elseif key == "PlaceholderText" then
                
            else
                pcall(function() input[key] = value end)
            end
        end,
    })
    return proxy
end

-- Makes every number of a saved build a whole number (rotation matrices only snap when already ~integer)
function normalizebuild(build)
	if type(build) ~= "table" then return build end
	for _, v in ipairs(build) do
		if type(v) == "table" and not v.isSign then
			local pp = v.p or v.pos
			if type(pp) == "table" then
				for i = 1, 3 do if tonumber(pp[i]) then local h = math.round(pp[i] * 2) / 2; if math.abs(pp[i] - h) < 0.02 then pp[i] = h end end end -- clean float noise only; keep .5 and keep off-grid blocks exact
				if #pp == 12 then
					for i = 4, 12 do
						local r = math.round(pp[i])
						if math.abs(pp[i] - r) < 0.001 then pp[i] = r end
					end
				end
			end
			local ss = v.s or v.size
			if type(ss) == "table" then
				for i = 1, 3 do if tonumber(ss[i]) then ss[i] = math.max(1, math.round(ss[i])) end end
			end
		end
	end
	return build
end

local function getGhostBlockers(ghost)
    -- Oriented box query against the physics spatial index: sees rotated parts, does not walk every brick,
    -- and is live on every call, so blocks placed a moment ago are caught too.
    local blockers = {}
    if not ghost or not ghost:IsA("BasePart") then return blockers end
    local bricks = workspace:FindFirstChild("Bricks")
    if not bricks then return blockers end
    local pad = 0.1 -- shrink the probe so parts that only touch a face are not counted
    local size = Vector3.new(math.max(ghost.Size.X - pad, 0.05), math.max(ghost.Size.Y - pad, 0.05), math.max(ghost.Size.Z - pad, 0.05))
    local op = OverlapParams.new()
    op.FilterType = Enum.RaycastFilterType.Include
    op.FilterDescendantsInstances = {bricks}
    op.MaxParts = 0
    local ok, parts = pcall(workspace.GetPartBoundsInBox, workspace, ghost.CFrame, size, op)
    if not ok then return blockers end
    for _, part in ipairs(parts) do
        if part ~= ghost and part.Parent and part.CanCollide == true then
            table.insert(blockers, part)
        end
    end
    return blockers
end

local function makeBlockerNonCollidable(part)
    if not part or not part.Parent or part.CanCollide == false then return true end
    local paint = (localplr.Character and localplr.Character:FindFirstChild("Paint")) or (localplr.Backpack and localplr.Backpack:FindFirstChild("Paint"))
    if not paint then return false end
    local ok = pcall(function()
        if not localplr.Character:FindFirstChild("Paint") then
            paint.Parent = localplr.Character
        end
        if localplr.Character:FindFirstChild("Paint") then
            local args = {
                [1] = part,
                [2] = Enum.NormalId.Top,
                [3] = part.Position + part.Size/2,
                [4] = "material",
                [5] = nil,
                [6] = "collide",
                [7] = ""
            }
            local event = (localplr.Character.Paint:FindFirstChild("origevent") and localplr.Character.Paint.origevent:Invoke(unpack(args))) or localplr.Character.Paint.Script.Event:FireServer(unpack(args))
            return true
        end
    end)
    if not ok then return false end
    local tries = 0
    repeat
        tries = tries + 1
        cwait(0.15)
        if not part.Parent then return true end
    until part.CanCollide == false or tries >= 12
    return part.CanCollide == false
end

local function clearCurrentGhostBlockers()

    local blockers = getGhostBlockers(oldprt)
    for _,part in ipairs(blockers) do
        makeBlockerNonCollidable(part)
    end
    return true
end




local function clearBlockersForSign(P, signSize, excludeParts)
    excludeParts = excludeParts or {}
    local pad = 0.05
    local probe = Instance.new("Part")
    probe.Anchored = true
    probe.CanCollide = false
    probe.CanQuery = false
    probe.CanTouch = false
    probe.CastShadow = false
    probe.Transparency = 1
    probe.Size = Vector3.new(signSize.X + pad, signSize.Y + pad, signSize.Z + pad)
    probe.CFrame = CFrame.new(P)
    probe.Parent = workspace

    local op = OverlapParams.new()
    op.FilterType = Enum.RaycastFilterType.Include
    op.FilterDescendantsInstances = {workspace.Bricks}
    op.MaxParts = 50

    local blockers = {}
    for _, p in ipairs(workspace:GetPartsInPart(probe, op)) do
        if p ~= probe and p.Parent and p.CanCollide == true and not excludeParts[p] then
            table.insert(blockers, p)
        end
    end
    probe:Destroy()

    if #blockers == 0 then return 0 end

    if localplr.Backpack:FindFirstChild("Paint") and localplr.Character then
        pcall(function() localplr.Backpack.Paint.Parent = localplr.Character end)
    end
    local paint = localplr.Character and localplr.Character:FindFirstChild("Paint")
    if not paint then return 0 end

    for _, blk in ipairs(blockers) do
        pcall(function()
            local args = { blk, Enum.NormalId.Top, blk.Position + blk.Size/2, "material", nil, "collide", "" }
            local ev = paint:FindFirstChild("origevent")
            if ev then ev:Invoke(unpack(args)); return end
            local scr = paint:FindFirstChild("Script")
            if scr and scr:FindFirstChild("Event") then scr.Event:FireServer(unpack(args)) end
        end)
        cwait(0.3)
    end

    return #blockers
end

local FACE_NORMALS = {
    Front  = Vector3.new(0, 0, -1),
    Back   = Vector3.new(0, 0, 1),
    Left   = Vector3.new(-1, 0, 0),
    Right  = Vector3.new(1, 0, 0),
    Top    = Vector3.new(0, 1, 0),
    Bottom = Vector3.new(0, -1, 0),
}

local function isAtGridXZ(pos)
    local function nearGrid(n)
        local k = math.round((n - 2) / 4)
        return math.abs(n - (k * 4 + 2)) < 0.1
    end
    return nearGrid(pos.X) and nearGrid(pos.Z)
end

local function computeBlockPlacement(P, faceName, signSize)
    local n = FACE_NORMALS[faceName] or Vector3.new(0, 0, -1)
    local sx, sy, sz = signSize.X, signSize.Y, signSize.Z
    local signThick = math.min(sx, sy, sz)
    local signY = sy
    local signWidth
    if math.abs(sx - signThick) < 0.01 then
        signWidth = sz
    elseif math.abs(sz - signThick) < 0.01 then
        signWidth = sx
    else
        signWidth = math.max(sx, sz)
    end
    local blockHeight
    if signY >= 1 then
        blockHeight = signY + 1
    else
        blockHeight = 1
    end
    local blockDepth = signWidth
    local blockSize
    if faceName == "Front" or faceName == "Back" then
        blockSize = Vector3.new(signWidth, blockHeight, blockDepth)
    elseif faceName == "Left" or faceName == "Right" then
        blockSize = Vector3.new(blockDepth, blockHeight, signWidth)
    else
        blockSize = Vector3.new(signWidth, blockHeight, blockDepth)
    end
    local blockCenter = P - n * (signThick / 2 + blockDepth / 2)
    return blockCenter, blockSize
end

local function applySignColorMaterial(signPart, signData, materialsTable)
    if not signPart or not signPart.Parent then return end
    if not signPart:FindFirstChildOfClass("SurfaceGui") then return end

    local wantColor = nil
    if type(signData.c) == "table" and #signData.c == 3 then
        wantColor = Color3.fromRGB(signData.c[1], signData.c[2], signData.c[3])
    end
    local wantMatName = nil
    if signData.o and materialsTable then
        local ok, mat = pcall(function() return Enum.Material[signData.o] end)
        if ok and mat and materialsTable[mat] then
            wantMatName = materialsTable[mat]
        end
    end
    if not wantColor and not wantMatName then return end

    if localplr.Backpack:FindFirstChild("Paint") and localplr.Character then
        pcall(function() localplr.Backpack.Paint.Parent = localplr.Character end)
    end
    if not (localplr.Character and localplr.Character:FindFirstChild("Paint")) then return end

    local faceName = signData.signFace or "Front"
    local faceId   = Enum.NormalId[faceName] or Enum.NormalId.Front
    local n        = FACE_NORMALS[faceName] or Vector3.new(0,0,-1)

    local surfacePos = signPart.Position + Vector3.new(
        n.X * signPart.Size.X / 2,
        n.Y * signPart.Size.Y / 2,
        n.Z * signPart.Size.Z / 2
    )

    local function firePaint(args)
        pcall(function()
            local p = localplr.Character.Paint
            local ev = p:FindFirstChild("origevent")
            if ev then ev:Invoke(unpack(args)); return end
            local scr = p:FindFirstChild("Script")
            if scr and scr:FindFirstChild("Event") then scr.Event:FireServer(unpack(args)) end
        end)
    end

    if wantColor and wantColor ~= signPart.Color then
        local c = 0
        repeat
            c = c + 1
            if localplr.Character and not localplr.Character:FindFirstChild("Paint")
               and localplr.Backpack:FindFirstChild("Paint") then
                localplr.Backpack.Paint.Parent = localplr.Character
            end
            firePaint({ signPart, faceId, surfacePos, "color", wantColor, "tiles", "" })
            swait(0.3)
        until not signPart.Parent or signPart.Color == wantColor or c > 20
    end

    if wantMatName and (not materialsTable or materialsTable[signPart.Material] ~= wantMatName) then
        local c = 0
        repeat
            c = c + 1
            if localplr.Character and not localplr.Character:FindFirstChild("Paint")
               and localplr.Backpack:FindFirstChild("Paint") then
                localplr.Backpack.Paint.Parent = localplr.Character
            end
            firePaint({ signPart, faceId, surfacePos, "material", nil, wantMatName, "" })
            swait(0.3)
        until not signPart.Parent
              or (materialsTable and materialsTable[signPart.Material] == wantMatName)
              or c > 20
    end
end

local function placeSign(signData, parentPart, materialsTable)
	local raw
	if signData._worldPos then raw = {signData._worldPos.X, signData._worldPos.Y, signData._worldPos.Z}
	else raw = signData.p or signData.pos end
	if not raw then return nil end

	local P
	if #raw == 12 then
		P = CFrame.new(table.unpack(raw)).Position
	else
		P = Vector3.new(raw[1], raw[2], raw[3])
		if signData.s and not signData._worldPos then
			local size = Vector3.new(table.unpack(signData.s))
			P = Vector3.new(
				P.X + size.X/2 - 0.5,
				P.Y + size.Y/2 - 0.5,
				P.Z + size.Z/2 - 0.5
			)
		end
	end

	local faceName = signData.signFace or "Front"
	local faceId   = Enum.NormalId[faceName] or Enum.NormalId.Front
	local signType = signData.signType or "wall"
	local signText = signData.signText or ""
	local wantCollide = signData.cc
	if wantCollide == nil then wantCollide = signData.collide end

	local n = FACE_NORMALS[faceName] or Vector3.new(0,0,-1)

	local sw, sh, st = 4, 3, 0.5
	if signData.signSize then
		sw = tonumber(signData.signSize[1]) or 4
		sh = tonumber(signData.signSize[2]) or 3
		st = tonumber(signData.signSize[3]) or 0.5
	end
	local signThick = math.min(sw, sh, st)
	local blockFacePos = P - n * (signThick / 2)

	if localplr.Backpack:FindFirstChild("Sign") and localplr.Character then
		pcall(function() localplr.Backpack.Sign.Parent = localplr.Character end)
	end
	if not (localplr.Character and localplr.Character:FindFirstChild("Sign")) then
		notify("Sign", "No Sign tool in Backpack", "Warning")
		return nil
	end
	local sign = localplr.Character.Sign

	local function teleport(viewPos, target)
		if not (localplr.Character and localplr.Character:FindFirstChild("HumanoidRootPart")) then return end
		pcall(function()
			lib.markTP(); localplr.Character.HumanoidRootPart.CFrame = CFrame.lookAt(viewPos, target)
		end)
		swait(0.45)
	end

	local function findNearbySign(maxDist, exclude)
		maxDist = maxDist or 5
		exclude = exclude or {}
		local myModel = workspace.Bricks:FindFirstChild(localplr.Name)
		if not myModel then return nil end
		local best, bestDist = nil, maxDist
		for _, p in ipairs(myModel:GetChildren()) do
			if p:IsA("BasePart") and not exclude[p] then
				local sg = p:FindFirstChildOfClass("SurfaceGui")
				if sg and sg:FindFirstChildWhichIsA("TextBox", true) then
					local d = (p.Position - P).Magnitude
					if d < bestDist then best = p; bestDist = d end
				end
			end
		end
		return best
	end

	local function applySignText(signPart, text)
		if not signPart or not text or text == "" then return end
		local sg = signPart:FindFirstChildOfClass("SurfaceGui")
		if not sg then return end
		local tb = sg:FindFirstChildWhichIsA("TextBox", true)
		if not tb then return end
		pcall(function()
			tb.Text = tostring(text)
			tb:CaptureFocus()
			task.wait(0.15 * SIGN_WAIT_MULT)
			tb:ReleaseFocus(true)
		end)
	end

	local function deletePartViaTool(part)
		if not part or not part.Parent then return end
		local del = localplr.Backpack:FindFirstChild("Delete") or (localplr.Character and localplr.Character:FindFirstChild("Delete"))
		if not del then return end
		if localplr.Character and not localplr.Character:FindFirstChild("Delete") then
			pcall(function() del.Parent = localplr.Character end)
		end
		local d2 = localplr.Character and localplr.Character:FindFirstChild("Delete")
		if not d2 then return end
		pcall(function()
			if d2:FindFirstChild("origevent") then
				d2.origevent:Invoke(part, part.Position)
			elseif d2:FindFirstChild("Script") and d2.Script:FindFirstChild("Event") then
				d2.Script.Event:FireServer(part, part.Position)
			end
		end)
	end

	local placedSign = nil

	if signType == "free" then
		local lookDir = n
		if signData.p and #signData.p == 12 then
			local cf = CFrame.new(table.unpack(signData.p))
			local flat = Vector3.new(cf.LookVector.X, 0, cf.LookVector.Z)
			if flat.Magnitude > 0.01 then lookDir = flat.Unit end
		end
		local viewPos = P + lookDir * 6 + Vector3.new(0, 1, 0)
		teleport(viewPos, P)

		clearBlockersForSign(P, Vector3.new(sw, sh, st), {})

		local rayOrigin = P - Vector3.new(0, 2, 0)
		local pp = RaycastParams.new()
		pp.FilterType = Enum.RaycastFilterType.Include
		pp.FilterDescendantsInstances = {workspace.Terrain, workspace.Bricks}
		local hit = workspace:Raycast(rayOrigin, Vector3.new(0,-50,0), pp)
		local groundPos = hit and hit.Position or (P - Vector3.new(0, 2, 0))

		local args = { workspace.Terrain, Enum.NormalId.Top, groundPos, "normal" }
		pcall(function()
			if sign:FindFirstChild("origevent") then sign.origevent:Invoke(unpack(args))
			elseif sign:FindFirstChild("Script") and sign.Script:FindFirstChild("Event") then sign.Script.Event:FireServer(unpack(args)) end
		end)

		if placedSign and signText ~= "" then
			applySignText(placedSign, signText)
			swait(0.4)
		end

			if myModel then
				local toDelete = {}

					if p:IsA("BasePart") and p ~= placedSign then
						local hasGui = false
						for _, sg in ipairs(p:GetChildren()) do
							if sg:IsA("SurfaceGui") then hasGui = true; break end
						end
						if not hasGui then
							local dx = math.abs(p.Position.X - placedSign.Position.X)
							local dz = math.abs(p.Position.Z - placedSign.Position.Z)
							local dy = placedSign.Position.Y - p.Position.Y
							if dx < 0.3 and dz < 0.3 and dy > 0.5 and dy < 3.5 then
								table.insert(toDelete, p)
							end
						end
					end
				end
				for _, p in ipairs(toDelete) do
					deletePartViaTool(p)
					swait(0.4)
				end
			end
		end
	else
		if parentPart and parentPart.Parent then
			local viewPos = P + n * 6 + Vector3.new(0, 1, 0)
			teleport(viewPos, P)
			swait(0.2)
			clearBlockersForSign(P, Vector3.new(sw, sh, st), {parentPart})
			local args = { parentPart, faceId, blockFacePos, "normal" }
			pcall(function()
				if sign:FindFirstChild("origevent") then sign.origevent:Invoke(unpack(args))
				elseif sign:FindFirstChild("Script") and sign.Script:FindFirstChild("Event") then sign.Script.Event:FireServer(unpack(args)) end
			end)
			swait(0.7)
			placedSign = findNearbySign(3)
			if placedSign and signText ~= "" then
				applySignText(placedSign, signText)
				swait(0.4)
			end
		else
			local viewPos = P + n * 6 + Vector3.new(0, 1, 0)
			teleport(viewPos, P)

			clearBlockersForSign(P, Vector3.new(sw, sh, st), {})

			local blockCenter, blockSize = computeBlockPlacement(P, faceName, Vector3.new(sw, sh, st))
			local tempBlock = buildTempBlock(blockCenter, blockSize)
			swait(0.5)

			if not tempBlock then
				notify("Sign", "Could not build temp block", "Error")
				if oldprt then oldprt:Destroy(); oldprt = nil end
				return nil
			end

			local args = { tempBlock, faceId, blockFacePos, "normal" }
			pcall(function()
				if sign:FindFirstChild("origevent") then sign.origevent:Invoke(unpack(args))
				elseif sign:FindFirstChild("Script") and sign.Script:FindFirstChild("Event") then sign.Script.Event:FireServer(unpack(args)) end
			end)
			swait(0.7)
			placedSign = findNearbySign(3)
			if placedSign and signText ~= "" then
				applySignText(placedSign, signText)
				swait(0.4)
			end

			for _ = 1, 12 do
				if not tempBlock or not tempBlock.Parent then break end
				deletePartViaTool(tempBlock)
				swait(0.5)
			end
			if oldprt then oldprt:Destroy(); oldprt = nil end
		end
	end

	if placedSign and placedSign.Parent and wantCollide == false then
		local paint = localplr.Backpack:FindFirstChild("Paint") or (localplr.Character and localplr.Character:FindFirstChild("Paint"))
		if paint then
			if not localplr.Character:FindFirstChild("Paint") then
				pcall(function() paint.Parent = localplr.Character end)
			end
			local p = localplr.Character:FindFirstChild("Paint")
			if p then
				local args = { placedSign, Enum.NormalId.Top, placedSign.Position + placedSign.Size/2, "material", nil, "collide", "" }
				pcall(function()
					local ev = p:FindFirstChild("origevent")
					if ev then ev:Invoke(unpack(args)); return end
					local scr = p:FindFirstChild("Script")
					if scr and scr:FindFirstChild("Event") then scr.Event:FireServer(unpack(args)) end
				end)
				swait(0.3)
			end
		end
	end

	if placedSign and placedSign.Parent then
		applySignColorMaterial(placedSign, signData, materialsTable)
	end

	return placedSign
end

function createpartpos(pos,col)
	if typeof(pos) == "CFrame" then
		pos = pos.Position
	end
	local p = Instance.new("Part")
	p.Shape = Enum.PartType.Ball
	p.Anchored = true
	p.CanCollide = false
	p.CastShadow = false
	p.CanQuery = false
	p.Color = col

function createpartrepl(pos,bsize,col,mat,transp,anch,collide,sprays)
	if typeof(pos) == "Vector3" then
		pos = CFrame.new(pos)
	end
	local p = Instance.new("Part")
	oldprt = p
	p.Anchored = anch or true
	p.CanCollide = collide
	if p.CanCollide == nil then
		p.CanCollide = false
	end
	p.CastShadow = false
	p.CanQuery = false
	p.Color = col
	p.Transparency = transp or .5
	p.Material = mat
	if bsize ~= nil then
		pos = CFrame.new((pos.X + (bsize.X/2))-.5,(pos.Y + (bsize.Y/2))-.5,(pos.Z + (bsize.Z/2))-.5) * pos.Rotation
	end
	p.Size = bsize or Vector3.new(mult,mult,mult)
	p.CFrame = pos
	if sprays then
		for i,v in pairs(sprays) do
			local face = Enum.NormalId[v[1]]
			local image = v[2]
			local txt = v[3]
			local surfaceui = Instance.new("SurfaceGui")
			surfaceui.Face = face
			surfaceui.SizingMode = Enum.SurfaceGuiSizingMode.PixelsPerStud
			surfaceui.PixelsPerStud = 50
			local out,count = string.gsub(txt,"#","l")
			if count == string.len(txt) then
				local img = Instance.new("ImageLabel")
				img.Image = image
				img.BackgroundTransparency = 1
				img.Size = UDim2.new(1,0,1,0)
				img.Parent = surfaceui
			else
				local textlabel = Instance.new("TextLabel")
				textlabel.Text = txt
				textlabel.BackgroundTransparency = 1
				textlabel.TextScaled = true
				textlabel.TextColor3 = Color3.fromRGB(255,255,255)
				textlabel.Font = Enum.Font.FredokaOne
				textlabel.Size = UDim2.new(1,0,1,0)
				textlabel.Parent = surfaceui
			end
			surfaceui.Parent = p
		end
	end
	p.Parent = workspace
	return p
end

function saveblock(bl)
	local blockdata = {}
	if bl:IsA("BasePart") then
		local p = bl.Position
		local pt = {bl.Position.X,bl.Position.Y,bl.Position.Z}
		if (bl.CFrame - bl.Position) ~= CFrame.new() then
			blockdata.p = {bl.CFrame:GetComponents()}
		else
			blockdata.p = pt
		end
		blockdata.c = {math.round(bl.Color.R*255),math.round(bl.Color.G*255),math.round(bl.Color.B*255)}
		blockdata.a = bl.Anchored
		blockdata.cc = bl.CanCollide
		if bl.Size.X ~= mult or bl.Size.Y ~= mult or bl.Size.Z ~= mult then
			blockdata.p[1] = (blockdata.p[1] - (bl.Size.X/2))+.5
			blockdata.p[2] = (blockdata.p[2] - (bl.Size.Y/2))+.5
			blockdata.p[3] = (blockdata.p[3] - (bl.Size.Z/2))+.5
			blockdata.s = {bl.Size.X,bl.Size.Y,bl.Size.Z}
		else
		end
		for i = 1, 3 do local h = math.round(blockdata.p[i] * 2) / 2; if math.abs(blockdata.p[i] - h) < 0.02 then blockdata.p[i] = h end end -- clean float noise only; keep .5 and keep off-grid blocks exact
		if #blockdata.p == 12 then
			for i = 4, 12 do
				local r = math.round(blockdata.p[i])
				if math.abs(blockdata.p[i] - r) < 0.001 then blockdata.p[i] = r end
			end
		end
		if blockdata.s then
			for i = 1, 3 do blockdata.s[i] = math.max(1, math.round(blockdata.s[i])) end
		end
		blockdata.m = materials[bl.Material] or "plastic"
		blockdata.o = bl.Material.Name
		blockdata.sp = {}
		for i,v in pairs(bl:GetChildren()) do
			if v.Name == "Spray" then
				local escaped = (v.Label.Text:gsub('"','\"'))
				table.insert(blockdata.sp,{v.Face.Name,v.Image.Image,escaped})
			end
		end
	end
	return blockdata
end

isSignPanel = function(part)
	if not part:IsA("BasePart") then return false, nil, nil end
	for _, sg in ipairs(part:GetChildren()) do
		if sg:IsA("SurfaceGui") then
			local tb = sg:FindFirstChildWhichIsA("TextBox", true)
			if tb then return true, sg.Face.Name, tb.Text or "" end
		end
	end
	return false, nil, nil
end

findSignLeg = function(panel, model)
	for _, p in ipairs(model:GetChildren()) do
		if p:IsA("BasePart") and p ~= panel then
			local hasGui = false
			for _, sg in ipairs(p:GetChildren()) do if sg:IsA("SurfaceGui") then hasGui = true; break end end
			if not hasGui then
				local dx = math.abs(p.Position.X - panel.Position.X)
				local dz = math.abs(p.Position.Z - panel.Position.Z)
				local dy = panel.Position.Y - p.Position.Y
				if dx < 0.1 and dz < 0.1 and dy > 0.5 and dy < 3.5 and p.Size.Y <= 2.5 then return p end
			end
		end
	end
	return nil
end

findParentPart = function(panel, model)
	if not panel or not panel.Parent then return nil end
	local ok, face = isSignPanel(panel)
	if not ok then return nil end
	local n = FACE_NORMALS[face] or Vector3.new(0,0,-1)
	local signThick = math.min(panel.Size.X, panel.Size.Y, panel.Size.Z)
	local contactPos = panel.Position - n * (signThick / 2)
	for _, p in ipairs(model:GetChildren()) do
		if p:IsA("BasePart") and p ~= panel then
			local rel = p.CFrame:PointToObjectSpace(contactPos)
			local h = p.Size / 2
			if math.abs(rel.X) <= h.X + 0.05 and math.abs(rel.Y) <= h.Y + 0.05 and math.abs(rel.Z) <= h.Z + 0.05 then
				return p
			end
		end
	end
	return nil
end

-- UserId lookup for players who already left (cached, retried after 60s on failure)
local function lookupId(name)
	local rec = idlookup[name]
	if rec then
		if rec.id then return rec.id end
		if rec.pending or os.clock() - rec.t < 60 then return nil end
	end
	rec = {pending = true, t = os.clock()}
	idlookup[name] = rec
	task.spawn(function()
		local ok, id = pcall(function() return game.Players:GetUserIdFromNameAsync(name) end)
		if ok and type(id) == "number" then
			rec.id = id
		else
			rec.pending = false
			rec.t = os.clock()
		end
	end)
	return nil
end
local function scanPlayerBuilds()
	local entries = {}
	for _, p in ipairs(game.Players:GetPlayers()) do remember(p) end
	for _, model in ipairs(workspace.Bricks:GetChildren()) do
		if model:IsA("Model") or model:IsA("Folder") then
			local count = countBricksInModel(model)
			if count > 0 then
				local plr = game.Players:FindFirstChild(model.Name)
				local ingame = plr ~= nil and plr:IsA("Player")
				local info = seenplrs[model.Name]
				local uid = (info and info.UserId) or lookupId(model.Name)
				table.insert(entries,{Key=model.Name,UserId=uid,Display=info and info.Display,Name=model.Name,Count=count,InGame=ingame})
			end
		end
	end
	table.sort(entries,function(a,b)
		if a.InGame ~= b.InGame then return a.InGame end
		if a.Count ~= b.Count then return a.Count > b.Count end
		return a.Name:lower() < b.Name:lower()
	end)
	if plrlist then
		plrlist.SetEntries(entries)
		local key = plrlist.Get()
		if key then plrbuild = workspace.Bricks:FindFirstChild(key) end
		refreshplrhighlight()
	end

-- Rotation helpers (fields on lib so the main chunk's 200-local limit is untouched).
-- A quarter turn only maps the block grid onto itself when it turns about a point whose X/Z are even and
-- equal mod 4 (blocks sit on 2 + 4n). So the pivot is the build centre snapped to that lattice; preview and
-- Load Save both use this same pivot, so what you see is exactly what gets built.
function lib.rotPivot()
	-- Pivot = build centre snapped to the lattice a quarter turn maps onto itself: all three coordinates
	-- even and equal mod 4 (blocks sit on 2 + 4n). Needed for X/Z turns too, so Y is snapped as well.
	local c = getPreviewCenter() or gizmoPivotCenter or Vector3.new(0,0,0)
	local best, bestD
	for _, r in ipairs({0, 2}) do
		local function q(n) return math.round((n - r) / 4) * 4 + r end
		local pt = Vector3.new(q(c.X), q(c.Y), q(c.Z))
		local d = (pt - c).Magnitude
		if not bestD or d < bestD then best, bestD = pt, d end
	end
	return best
end
lib.rotationX, lib.rotationZ = 0, 0
function lib.rotationCF() -- Z, then X (tilt), then Y (spin about world up); every angle is a multiple of 90
	local function q(d) return math.rad(math.floor(((tonumber(d) or 0) % 360) / 90 + 0.5) * 90) end
	return CFrame.Angles(0, q(lib.rotationDegrees), 0) * CFrame.Angles(q(lib.rotationX), 0, q(lib.rotationZ))
end
function lib.refreshRotation()
	if prttable and #prttable > 0 then
		lib.applyPositionOffsetToPreview()
		createAxisGizmo()
	end
end
function lib.snapNum(n) -- clean float noise left by the rotation maths (keeps real .5 / off-grid values)
	local h = math.round(n * 2) / 2
	if math.abs(n - h) < 0.001 then return h end
	return n
end

function optimizeLoadOrder(build)
	if not build or #build < 2 then
		return build
	end

	local nodes = {}
	local groups = {}

	local function getpos(v)
		local p = v.p or v.pos
		if not p then return nil end
		return Vector3.new(p[1],p[2],p[3])
	end

	local function getsize(v)
		local size = v.s or v.size
		if size then
			return Vector3.new(size[1],size[2],size[3])
		end
		return Vector3.new(mult,mult,mult)
	end

	local function numkey(n)
		return tostring(math.round(n * 10000) / 10000)
	end

	local function poskey(p)
		return numkey(p.X).."|"..numkey(p.Y).."|"..numkey(p.Z)
	end

	local function sizekey(size)
		return numkey(size.X).."|"..numkey(size.Y).."|"..numkey(size.Z)
	end

	for i,v in ipairs(build) do
		local pos = getpos(v)
		if pos then
			local size = getsize(v)
			local node = {data=v,index=i,pos=pos,size=size}
			table.insert(nodes,node)
			local sk = sizekey(size)
			groups[sk] = groups[sk] or {}
			groups[sk][poskey(pos)] = node
		end
	end

	local dirs = {
		Vector3.new(1,0,0),Vector3.new(-1,0,0),
		Vector3.new(0,1,0),Vector3.new(0,-1,0),
		Vector3.new(0,0,1),Vector3.new(0,0,-1)
	}

	local ordered = {}
	local visited = {}

	for _,node in ipairs(nodes) do
		if not visited[node] then
			local queue = {node}
			local qhead = 1
			visited[node] = true
			local group = groups[sizekey(node.size)]

			while qhead <= #queue do
				local current = queue[qhead]
				qhead = qhead + 1
				table.insert(ordered,current.data)

				for _,dir in ipairs(dirs) do
					local np = current.pos + Vector3.new(
						dir.X * current.size.X,
						dir.Y * current.size.Y,
						dir.Z * current.size.Z
					)
					local nextnode = group[poskey(np)]
					if nextnode and not visited[nextnode] then
						visited[nextnode] = true
						table.insert(queue,nextnode)
					end
				end
			end
		end
	end

	return ordered
end

updateTransformFields = function()
	if updatingTransformFields then return end
	updatingTransformFields = true

	local ok, err = pcall(function()
		local base = intvec3(gizmoPivotCenter or getPreviewCenter() or Vector3.new(0,0,0))

end

-- Build position = exact centre of the build's bounding box as it is saved/loaded (NOT rounded),
-- same value the preview parts give at offset 0. The build is only ever moved by whole-stud offsets.
getPreviewCenter = function()
	local build = selectedbuild and selectedbuild[2]
	if type(build) ~= "table" then return nil end
	local function scan(includeSigns)
		local mn = Vector3.new(math.huge, math.huge, math.huge)
		local mx = Vector3.new(-math.huge, -math.huge, -math.huge)
		local found = false
		for _, v in ipairs(build) do
			if type(v) == "table" and (includeSigns or not v.isSign) then
				local pp = v.p or v.pos
				if type(pp) == "table" and tonumber(pp[1]) and tonumber(pp[2]) and tonumber(pp[3]) then
					local ss = v.s or v.size
					local size, c
					if type(ss) == "table" then
						size = Vector3.new(ss[1], ss[2], ss[3])
						c = Vector3.new(pp[1] + size.X/2 - 0.5, pp[2] + size.Y/2 - 0.5, pp[3] + size.Z/2 - 0.5)
					else
						size = Vector3.new(mult, mult, mult)
						c = Vector3.new(pp[1], pp[2], pp[3])
					end
					found = true
					local h = size / 2
					mn = Vector3.new(math.min(mn.X, c.X - h.X), math.min(mn.Y, c.Y - h.Y), math.min(mn.Z, c.Z - h.Z))
					mx = Vector3.new(math.max(mx.X, c.X + h.X), math.max(mx.Y, c.Y + h.Y), math.max(mx.Z, c.Z + h.Z))
				end
			end
		end
		if found then return (mn + mx) / 2 end
		return nil
	end
	return scan(false) or scan(true)
end

function lib.applyPositionOffsetToPreview()
	gizmoOffset = intvec3(gizmoOffset) or Vector3.new(0,0,0)
	if prttable and #prttable>0 then
		local pivot = lib.rotPivot()
		local transform = CFrame.new(pivot + gizmoOffset) * lib.rotationCF() * CFrame.new(-pivot)
		for i,v in ipairs(prttable) do
			local base=previewBaseCFrames[i]
			if v and v.Parent and base then
				-- The CFrame already carries the rotation: do NOT also swap Size (that rotated the part twice).
				v.CFrame=transform*base
			end
		end
		if offsetgizmo and offsetgizmo.anchor and offsetgizmo.anchor.Parent then
			offsetgizmo.anchor.CFrame=CFrame.new(offsetgizmo.base+gizmoOffset)
		end
	end
	updateTransformFields()
end

local function setPositionAxis(axis,value)
	if lib.positionLocked then updateTransformFields() return end
	value=tonumber(value)
	if not value or value ~= value or value == math.huge or value == -math.huge then return end
	value = math.round(value)
	local base = intvec3(gizmoPivotCenter or getPreviewCenter() or Vector3.new(0,0,0))
	local off = gizmoOffset
	-- shown position = whole-number reference (rounded load position) + whole-stud offset
	if axis=="X" then off=Vector3.new(value-base.X,off.Y,off.Z)
	elseif axis=="Y" then off=Vector3.new(off.X,value-base.Y,off.Z)
	else off=Vector3.new(off.X,off.Y,value-base.Z) end
	gizmoOffset=off
	lib.applyPositionOffsetToPreview()
end

-- Move gizmo: Roblox's built-in arrow Handles on the faces of the build's bounding box.
-- Dragging snaps to whole studs (1 stud steps) and works with mouse and touch.
createAxisGizmo = function()
	destroyOffsetGizmo()
	if not prttable or #prttable==0 then return end
	local minv=Vector3.new(math.huge,math.huge,math.huge)
	local maxv=Vector3.new(-math.huge,-math.huge,-math.huge)
	local valid=false
	for _,v in ipairs(prttable) do
		if v and v.Parent and v:IsA("BasePart") then
			valid=true
			local cfv,sz=v.CFrame,v.Size
			local rv,uv,lv=cfv.RightVector,cfv.UpVector,cfv.LookVector
			local half=Vector3.new(
				math.abs(rv.X)*sz.X+math.abs(uv.X)*sz.Y+math.abs(lv.X)*sz.Z,
				math.abs(rv.Y)*sz.X+math.abs(uv.Y)*sz.Y+math.abs(lv.Y)*sz.Z,
				math.abs(rv.Z)*sz.X+math.abs(uv.Z)*sz.Y+math.abs(lv.Z)*sz.Z)/2
			local p=v.Position
			minv=Vector3.new(math.min(minv.X,p.X-half.X),math.min(minv.Y,p.Y-half.Y),math.min(minv.Z,p.Z-half.Z))
			maxv=Vector3.new(math.max(maxv.X,p.X+half.X),math.max(maxv.Y,p.Y+half.Y),math.max(maxv.Z,p.Z+half.Z))
		end
	end
	if not valid then return end
	local center=(minv+maxv)/2
	local anchor=Instance.new("Part")
	anchor.Name="BuildMoveHandlesAnchor"
	anchor.Anchored=true
	anchor.CanCollide=false
	anchor.CanQuery=false
	anchor.CanTouch=false
	anchor.CastShadow=false
	anchor.Transparency=1
	anchor.Size=Vector3.new(math.max(maxv.X-minv.X,4),math.max(maxv.Y-minv.Y,4),math.max(maxv.Z-minv.Z,4))
	anchor.CFrame=CFrame.new(center)
	anchor.Parent=workspace.CurrentCamera or workspace
	local objects={anchor}
	local function addHandles(color,faces)
		local h=Instance.new("Handles")
		h.Adornee=anchor
		h.Style=Enum.HandlesStyle.Movement
		h.Faces=Faces.new(table.unpack(faces))
		h.Color3=color
		h.Visible=not lib.positionLocked
		h.Parent=game:GetService("CoreGui")
		objects[#objects+1]=h
		local startOffset=nil
		h.MouseButton1Down:Connect(function() if lib.positionLocked then return end startOffset=gizmoOffset end)
		h.MouseDrag:Connect(function(face,distance)
			if lib.positionLocked or not startOffset then return end
			gizmoOffset=startOffset+Vector3.FromNormalId(face)*math.round(distance)
			lib.applyPositionOffsetToPreview()
		end)
		h.MouseButton1Up:Connect(function() startOffset=nil end)
	end
	addHandles(Color3.fromRGB(235,85,85),{Enum.NormalId.Right,Enum.NormalId.Left})
	addHandles(Color3.fromRGB(95,215,120),{Enum.NormalId.Top,Enum.NormalId.Bottom})
	addHandles(Color3.fromRGB(100,150,235),{Enum.NormalId.Front,Enum.NormalId.Back})
	offsetgizmo={anchor=anchor,objects=objects,base=center-gizmoOffset}
end

function lib.moveBuildToPlayer() -- puts the preview in front of you, its bottom level with your feet
	if lib.positionLocked then notify("Transform", "Position is locked", "Warning") return end
	if not prttable or #prttable == 0 then notify("Transform", "Load a build preview first", "Warning") return end
	local char = localplr.Character
	local hrp = char and char:FindFirstChild("HumanoidRootPart")
	local hum = char and char:FindFirstChildOfClass("Humanoid")
	if not hrp then notify("Transform", "No character", "Warning") return end

	-- world bounds of the preview as it is right now (rotation and offset included)
	local mn = Vector3.new(math.huge, math.huge, math.huge)
	local mx = Vector3.new(-math.huge, -math.huge, -math.huge)
	for _, part in ipairs(prttable) do
		if part and part.Parent then
			local h = part.Size / 2
			for _, sx in ipairs({-1, 1}) do for _, sy in ipairs({-1, 1}) do for _, sz in ipairs({-1, 1}) do
				local c = part.CFrame:PointToWorldSpace(Vector3.new(h.X * sx, h.Y * sy, h.Z * sz))
				mn = Vector3.new(math.min(mn.X, c.X), math.min(mn.Y, c.Y), math.min(mn.Z, c.Z))
				mx = Vector3.new(math.max(mx.X, c.X), math.max(mx.Y, c.Y), math.max(mx.Z, c.Z))
			end end end
		end
	end
	if mn.X == math.huge then return end
	local center, half = (mn + mx) / 2, (mx - mn) / 2

	local feetY = hrp.Position.Y - ((hum and hum.RigType == Enum.HumanoidRigType.R15) and (hrp.Size.Y / 2 + hum.HipHeight) or 3)
	local look = Vector3.new(hrp.CFrame.LookVector.X, 0, hrp.CFrame.LookVector.Z)
	look = look.Magnitude > 0.01 and look.Unit or Vector3.new(0, 0, -1)
	local right = Vector3.new(-look.Z, 0, look.X)
	local margin = 3 -- gap between you and the build, and clearance kept around other players

	local others = {}
	for _, p in ipairs(game.Players:GetPlayers()) do
		local r = p ~= localplr and p.Character and p.Character:FindFirstChild("HumanoidRootPart")
		if r then table.insert(others, r.Position) end
	end
	local function hitsPlayer(c, y) -- would a build centred at c (bottom at y) touch anyone?
		local lo = Vector3.new(c.X - half.X - margin, y - margin, c.Z - half.Z - margin)
		local hi = Vector3.new(c.X + half.X + margin, y + half.Y * 2 + margin, c.Z + half.Z + margin)
		for _, pos in ipairs(others) do
			if pos.X >= lo.X and pos.X <= hi.X and pos.Y >= lo.Y and pos.Y <= hi.Y and pos.Z >= lo.Z and pos.Z <= hi.Z then
				return true
			end
		end
		return false
	end

	local dirs = {look, right, -right, -look} -- front first, then the other sides
	local target
	for ring = 0, 2 do
		for _, d in ipairs(dirs) do
			local reach = math.abs(d.X) * half.X + math.abs(d.Z) * half.Z + margin + ring * 8
			local c = Vector3.new(hrp.Position.X + d.X * reach, 0, hrp.Position.Z + d.Z * reach)
			if not hitsPlayer(c, feetY) then target = c break end
		end
		if target then break end
	end
	if not target then notify("Transform", "No free spot: players are in the way", "Warning") return end

	local delta = Vector3.new(target.X - center.X, feetY - mn.Y, target.Z - center.Z)
	gizmoOffset = gizmoOffset + Vector3.new(math.round(delta.X), math.round(delta.Y), math.round(delta.Z))
	lib.applyPositionOffsetToPreview()
	createAxisGizmo()
end

function makegear(gearname,gearid,save,changename)
	deletegear(gearid)
	if changename == nil then
		changename = true
	end
	if changename then
	local foundcopy = false
	for i,v in pairs(currgears) do
		if v[3] == gearname then
			foundcopy = true
		end
	end
	if foundcopy then
		local n = 1
		local gearname2 = gearname
		repeat
			local foundcopy2 = false
			gearname2 = gearname.." "..tostring(n)
			for i,v in pairs(currgears) do
				if v[3] == gearname2 then
					foundcopy2 = true
				end
			end
			n = n + 1
			if not foundcopy2 then
				foundcopy = false
			end
			task.wait()
		until foundcopy == false
		gearname = gearname2
	end
	end
	local gbutton = lib.makebutton("Copy "..gearname,geartab,function()
		setclipboard(gearcopybefore..tostring(gearid))
	end)
	local gbutton2 = lib.makebutton("Say "..gearname,geartab,function()
		say(gearcopybefore..tostring(gearid),"system")
	end)
	if save then
		currgears[gearid] = {gbutton,gbutton2,gearname,gearid}
		savegears()
	end
end
function deletegear(gearid)
	if currgears[gearid] then
		local v = currgears[gearid]
		pcall(function()
			if type(v) == "table" then
				if type(v[1]) == "table" and v[1].Destroy then v[1]:Destroy() end
				if type(v[2]) == "table" and v[2].Destroy then v[2]:Destroy() end
			end
		end)
		currgears[gearid] = nil
		savegears()
	end
end

local function applyLightingAnti(name, enabled, property, forcedValue)
    local lighting = game:GetService("Lighting")
    local object = lighting:FindFirstChild(name)
    if not object then return end

    if enabled then
        if antiLightingSaved[object] == nil then
            local ok, original = pcall(function() return object[property] end)
            if ok then antiLightingSaved[object] = original end
        end
        pcall(function() object[property] = forcedValue end)
    elseif antiLightingSaved[object] ~= nil then
        local original = antiLightingSaved[object]
        pcall(function() object[property] = original end)
        antiLightingSaved[object] = nil
    end
end

local function applyAntiNoclip(character)
    if not anti.noclip or not character then return end
    for _, part in ipairs(character:GetDescendants()) do
        if part:IsA("BasePart") then
            pcall(function()
                local group = part.CollisionGroup
                if antiCollisionSaved[part] == nil then
                    antiCollisionSaved[part] = group
                end
                if group == "NoClip" then
                    part.CollisionGroup = "Default"
                end
            end)
        end
    end
end

do -- scoped Anti-Teleport: catches position jumps BEFORE the physics step
    -- Heartbeat runs after physics, so the CFrame it records is where physics legitimately left you.
    -- Stepped runs before physics: if the root already moved since the last Heartbeat, something wrote
    -- to it (a server/other script teleport). That is rolled back before physics can build on the jump.
    local RunService = game:GetService("RunService")
    local THRESHOLD = 30 -- studs; a bigger jump than this in one frame counts as a teleport
    local steppedConn, heartbeatConn, trackedChar, safeCF
    local function stopTracking()
        if steppedConn then steppedConn:Disconnect(); steppedConn = nil end
        if heartbeatConn then heartbeatConn:Disconnect(); heartbeatConn = nil end
        trackedChar, safeCF = nil, nil
    end
    local function ignoring() -- teleports caused by this script (build, ToSpawn, Anti Void, Anti Glitch)
        return os.clock() < math.max(anti.teleportIgnoreUntil or 0, lib.tpIgnoreUntil or 0)
    end
    lib.maketoggle("Anti Teleport", modtab, function(enabled)
        anti.teleport = enabled == true
        stopTracking()
        if not anti.teleport then return end
        heartbeatConn = RunService.Heartbeat:Connect(function()
            if not anti.teleport or scriptUnloading then stopTracking(); return end
            local character = localplr.Character
            local root = character and character:FindFirstChild("HumanoidRootPart")
            if not root then trackedChar, safeCF = nil, nil; return end
            trackedChar = character
            safeCF = root.CFrame
        end)
        steppedConn = RunService.Stepped:Connect(function()
            if not anti.teleport or scriptUnloading then stopTracking(); return end
            local character = localplr.Character
            local root = character and character:FindFirstChild("HumanoidRootPart")
            if not root or character ~= trackedChar or not safeCF then return end -- new character (respawn): re-baseline
            local current = root.CFrame
            if ignoring() then safeCF = current; return end
            if (current.Position - safeCF.Position).Magnitude > THRESHOLD then
                pcall(function()
                    root.CFrame = safeCF
                    root.AssemblyLinearVelocity = Vector3.zero
                    root.AssemblyAngularVelocity = Vector3.zero
                end)
            end
        end)
        table.insert(conn, heartbeatConn)
        table.insert(conn, steppedConn)
    end, false, "Rolls back sudden position jumps before physics runs. Ignores teleports made by this script (build, ToSpawn, Anti Void, Anti Glitch). Cannot stop respawns.")
end
